import re

test_settings = {
    'Theme': 'dark',
    'Notifications': 'enabled',
    'Volume_high': 'high'
}

def add_setting(settings, settings_tuple):
    # Converts settings to lowercase
    key, val = settings_tuple
    key = key.lower()
    val = val.lower()

    # Validate & update keys/values
    if key in settings:
        return f"Setting '{key}' already exists! Cannot add a new setting with this name."
    if key not in settings:
        settings[key] = val
        return f"Setting '{key}' added with value '{val}' successfully!"

def update_setting(settings, settings_tuple):
    # Converts settings to lowercase
    key, val = settings_tuple
    key = key.lower()
    val = val.lower()

    # Updates keys/values
    if key in settings:
        settings[key] = val
        return f"Setting '{key}' updated to '{val}' successfully!"

    if key not in settings:
        return f"Setting '{key}' does not exist! Cannot update a non-existing setting."

def delete_setting(settings, key):
    # Converts settings to lowercase
    key = key.lower()

    # Checks if key exists, if it exists it removes it
    if key in settings:
        settings.pop(key)
        return f"Setting '{key}' deleted successfully!"
    else:
        return "Setting not found!"

def view_settings(settings):
    # Checks if settings is empty else prints settings
    if settings == {}:
        return 'No settings available.'
    else:
        result = ""
        for key in settings.items():
            result += key[0].capitalize() + ": " + key[1] + "\n"
    return "Current User Settings:\n" + result

view_settings(test_settings)
print(view_settings(test_settings))
